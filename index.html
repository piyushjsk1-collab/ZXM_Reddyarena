<script>
 const phoneNumber = "+917073984744"; // Update this with client's WhatsApp number
 const message = encodeURIComponent("Hello Reddy Arena! I want to order sports equipment and claim the 10% discount.");
 const whatsappUrl = `https://wa.me/${phoneNumber}?text=${message}`;

 // Assign WhatsApp URL to all [data-wa] links
 document.querySelectorAll("[data-wa]").forEach(function(link) {
   link.href = whatsappUrl;
 });

 // Modal Popup Logic
 const autoModal = document.getElementById("autoModal");
 const modalCard = document.getElementById("modalCard");
 const modalCloseBtn = document.getElementById("modalCloseBtn");

 // Show Modal Automatically on page load
 window.addEventListener("DOMContentLoaded", function() {
   setTimeout(function() {
     autoModal.classList.add("active");
   }, 250);
 });

 // Click inside Modal Card -> Redirects to WhatsApp
 modalCard.addEventListener("click", function(e) {
   if (!e.target.closest("#modalCloseBtn")) {
     window.open(whatsappUrl, "_blank");
   }
 });

 // Close Modal on Button Click
 modalCloseBtn.addEventListener("click", function(e) {
   e.stopPropagation();
   autoModal.classList.remove("active");
 });

 // Close Modal on Overlay Click (Outside Card)
 autoModal.addEventListener("click", function(e) {
   if (e.target === autoModal) {
     autoModal.classList.remove("active");
   }
 });
</script>
